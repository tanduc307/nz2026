options(scipen = 999)

unique(df_long$SCFA) -> vec_SCFA

df_term_ls <- list()

for(i in 1:length(vec_SCFA)){

df_out_1 <- df_long |> subset(SCFA == vec_SCFA[i] & Termness == "term")

df_out_1 <- df_out_1 |> subset(value != 0)

a <- boxplot(df_out_1$value)

df_out_1 <- df_out_1[ which(!(df_out_1$value %in% a$out)) , ]

df_term_ls[[i]] <- df_out_1
}
df_term_ls_final <- do.call(rbind, df_term_ls)
df_preterm_ls <- list()

df_out_1 <- df_out_1[ which(!(df_out_1$value %in% a$out)) , ]

df_preterm_ls[[i]] <- df_out_1

print(a$out)
}
